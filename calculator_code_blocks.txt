"use client";

import { useState, useMemo } from "react";
import {
  TrendingUp,
  Calculator,
  Calendar,
  Percent,
  ShieldCheck,
  ArrowRight,
  Sparkles,
  DollarSign,
  Briefcase,
} from "lucide-react";

export function RoiCalculator() {
  // Phase 2: Dynamic state hooks for live calculation
  const [initialInvestment, setInitialInvestment] = useState<number>(50000);
  const [monthlyContribution, setMonthlyContribution] = useState<number>(2500);
  const [years, setYears] = useState<number>(5);
  const [annualRate, setAnnualRate] = useState<number>(18);

  // Dynamic compound interest formula
  const calculation = useMemo(() => {
    const r = annualRate / 100 / 12;
    const n = years * 12;

    const fvInitial = initialInvestment * Math.pow(1 + r, n);
    const fvMonthly = r > 0 ? monthlyContribution * ((Math.pow(1 + r, n) - 1) / r) : monthlyContribution * n;

    const totalValue = Math.round(fvInitial + fvMonthly);
    const totalPrincipal = Math.round(initialInvestment + monthlyContribution * n);
    const totalGains = Math.max(0, totalValue - totalPrincipal);
    const roiPercentage = totalPrincipal > 0 ? Math.round((totalGains / totalPrincipal) * 100) : 0;
    const multiplier = totalPrincipal > 0 ? (totalValue / totalPrincipal).toFixed(2) : "1.00";

    const principalShare = totalValue > 0 ? Math.round((totalPrincipal / totalValue) * 100) : 50;
    const gainsShare = 100 - principalShare;

    return {
      totalValue,
      totalPrincipal,
      totalGains,
      roiPercentage,
      multiplier,
      principalShare,
      gainsShare,
    };
  }, [initialInvestment, monthlyContribution, years, annualRate]);

  const initialPresets = [25000, 50000, 100000, 250000];
  const horizonPresets = [3, 5, 10, 15];

  return (
    <section id="calculator" className="scroll-mt-24 pt-24 pb-28 bg-white relative overflow-hidden border-t border-gray-100">
      <div className="absolute top-1/3 -right-40 w-96 h-96 bg-brand-primary/5 rounded-full blur-3xl pointer-events-none" />

        <div className="text-center max-w-3xl mx-auto mb-14">
          <div className="inline-flex items-center gap-2 px-3.5 py-1.5 rounded-full bg-brand-primary/10 text-brand-primary text-xs font-bold uppercase tracking-wider mb-4 border border-brand-primary/20">
            <Calculator size={14} />

            Interactive Investment & ROI Calculator
          </h2>
          <div className="w-16 h-1 bg-brand-accent mx-auto mt-4 rounded-full" />

                    ${initialInvestment.toLocaleString()}

  Calendar,
  Percent,
  ShieldCheck,
  ArrowRight,

  const [monthlyContribution, setMonthlyContribution] = useState<number>(2500);
  const [years, setYears] = useState<number>(5);
  const [annualRate, setAnnualRate] = useState<number>(18);

  const calculation = useMemo(() => {
    const r = annualRate / 100 / 12;
    const n = years * 12;

    const fvInitial = initialInvestment * Math.pow(1 + r, n);
    const fvMonthly = r > 0 ? monthlyContribution * ((Math.pow(1 + r, n) - 1) / r) : monthlyContribution * n;

    const totalValue = Math.round(fvInitial + fvMonthly);
    const totalPrincipal = Math.round(initialInvestment + monthlyContribution * n);
    const totalGains = Math.max(0, totalValue - totalPrincipal);
    const roiPercentage = totalPrincipal > 0 ? Math.round((totalGains / totalPrincipal) * 100) : 0;
    const multiplier = totalPrincipal > 0 ? (totalValue / totalPrincipal).toFixed(2) : "1.00";

    const principalShare = totalValue > 0 ? Math.round((totalPrincipal / totalValue) * 100) : 50;
    const gainsShare = 100 - principalShare;

    return {
      totalValue,
      totalPrincipal,
      totalGains,
      roiPercentage,
      multiplier,
      principalShare,
      gainsShare,
    };
  }, [initialInvestment, monthlyContribution, years, annualRate]);

  const horizonPresets = [3, 5, 10, 15];

              {/* Slider 1: Initial Investment */}

                <div className="flex items-center justify-between mb-2">

                  min={10000}
                  max={500000}
                  step={5000}

                          initialInvestment === val
                            ? "bg-brand-primary text-brand-accent shadow-xs"
                            : "bg-white hover:bg-gray-100 text-brand-dark/70 border border-gray-200"

                    </span>
                  </div>

                  <input
                    id="time-horizon"
                    type="range"
                    min={1}
                    max={20}
                    step={1}
                    value={years}
                    onChange={(e) => setYears(Number(e.target.value))}
                    className="w-full h-2.5 bg-gray-200 rounded-lg appearance-none cursor-pointer accent-brand-primary focus:outline-none"
                  />

                  <div className="flex gap-1.5 mt-2 justify-between">
                    {horizonPresets.map((y) => (
                      <button
                        key={y}
                        type="button"
                        onClick={() => setYears(y)}
                        className={`flex-1 py-1 rounded-md text-[11px] font-semibold transition-all cursor-pointer ${
                          years === y

                        }`}
                      >
                        {y}y
                      </button>
                    ))}
                  </div>
                </div>

                <div>
                  <div className="flex items-center justify-between mb-2">
                    <label htmlFor="projected-yield" className="flex items-center gap-1.5 text-xs sm:text-sm font-bold uppercase tracking-wider text-brand-dark/70">
                      <Percent size={15} className="text-brand-primary" />
                      <span>Target Yield Rate</span>
                    </label>
                    <span className="text-sm sm:text-base font-extrabold text-brand-primary bg-brand-primary/10 px-2.5 py-0.5 rounded-lg">
                      {annualRate}% p.a.
                    </span>
                  </div>

                  <input
                    id="projected-yield"
                    type="range"
                    min={8}
                    max={30}
                    step={1}
                    value={annualRate}
                    onChange={(e) => setAnnualRate(Number(e.target.value))}
                    className="w-full h-2.5 bg-gray-200 rounded-lg appearance-none cursor-pointer accent-brand-primary focus:outline-none"
                  />

                  <div className="flex items-center justify-between text-[11px] text-brand-dark/50 mt-2">
                    <span>8% (Conservative)</span>
                    <span className="font-semibold text-brand-primary">30% (Max)</span>
                  </div>

              </div>

              <ShieldCheck size={16} className="text-brand-primary shrink-0 mt-0.5" />
              <span>
                Calculations simulate monthly compounded investment returns typical of prime Nigerian civil infrastructure contracts and EPC asset delivery.
              </span>
            </div>
          </div>

          {/* Right Column: Animated Projected Results Display */}
          <div className="lg:col-span-5 bg-brand-secondary text-white rounded-3xl p-7 sm:p-9 shadow-xl border border-brand-primary/30 flex flex-col justify-between relative overflow-hidden">
            <div className="absolute top-0 right-0 w-64 h-64 bg-brand-accent/10 rounded-full blur-2xl pointer-events-none" />

            <div>
              <div className="flex items-center justify-between mb-6">
                <span className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-bold uppercase tracking-wider bg-brand-primary text-brand-accent border border-brand-accent/20">
                  <Sparkles size={13} />
                  <span>Projected Returns</span>
                </span>
                <span className="text-xs font-bold px-2.5 py-1 rounded-lg bg-white/10 text-brand-accent">
                  {calculation.multiplier}x Multiplier
                </span>
              </div>

              <div className="mb-6">
                <p className="text-xs font-bold uppercase tracking-wider text-white/60 mb-1">
                  Estimated Total Future Value
                </p>
                <motion.div
                  key={calculation.totalValue}
                  initial={{ scale: 0.96, opacity: 0.8 }}

                  <motion.div
                    initial={{ width: 0 }}
                    animate={{ width: `${calculation.gainsShare}%` }}
                    transition={{ type: "spring", stiffness: 80, damping: 15 }}
                    className="h-full bg-brand-accent"
                    title={`Compound Gains: $${calculation.totalGains.toLocaleString()}`}
                  />
                </div>

                    Total Invested
                  </p>
                  <p className="text-lg sm:text-xl font-bold text-white tracking-tight">
                    ${calculation.totalPrincipal.toLocaleString()}
                  </p>
                </div>
